function y = linear_convolution(x1, x2)    
    N = length(x1) + length(x2) - 1;
    x1_padded = [x1, zeros(1, N - length(x1))];
    x2_padded = [x2, zeros(1, N - length(x2))];
    X1 = fft(x1_padded);
    X2 = fft(x2_padded);
    Y = X1 .* X2;
    y = ifft(Y);
end
